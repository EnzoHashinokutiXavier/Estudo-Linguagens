
    Questões – INSERT, DELETE e UPDATE

Insira um novo aluno chamado “Ricardo Mendes”, nascido em 2001-08-12, da cidade de Dourados.
INSERT INTO alunos (nome, email, data_nascimento, id_cidade) VALUES ('Ricardo Mendes', 'ricardo@gmail.com', '2001-08-12, 1');


Cadastre um novo curso chamado “Machine Learning”, com carga horária de 90 horas e valor de R$ 1800,00.
INSERT INTO cursos (nome, carga_horaria, valor) VALUES ('Machine Learning', 90, 1800.00);


Atualize o valor do curso “Banco de Dados” para R$ 950,00.
UPDATE cursos SET valor = 950.00 WHERE nome = 'Banco de Dados';


Altere o status da matrícula de código 5 para “Ativa”.
UPDATE matriculas SET id_status = 1 WHERE id_matricula = 5;

DELETE FROM  cursos WHERE id_curso = 5;

    Questões – Consultas Simples 

Liste todos os alunos ordenados pelo nome em ordem alfabética.
SELECT * FROM alunos ORDER BY nome ASC;


Liste os cursos com valor maior que R$ 1000,00.
SELECT * FROM cursos WHERE valor > 1000.00;


Mostre os alunos nascidos entre 1999-01-01 e 2001-12-31.
SELECT * FROM alunos WHERE data_nascimento BETWEEN '199-01-01' AND '2001-12-31';


Liste AS matrículas utilizando alias para os nomes das colunas.
SELECT id_matricula AS codigo_de_matricula, id_aluno AS id_do_aluno FROM matriculas;


Liste os cursos ordenados do maior para o menor valor.
SELECT * FROM cursos ORDER BY valor DESC;


    Questões – Funções, GROUP BY e HAVING

Mostre a quantidade total de alunos cadastrados.
SELECT COUNT(*) FROM alunos;


Mostre o valor médio dos cursos.
SELECT AVG(valor) FROM cursos;


Mostre a quantidade de matrículas por status.
SELECT COUNT(*) FROM matriculas GROUP BY id_status;


Mostre a soma total dos valores dos cursos agrupados pela carga horária.
SELECT SUM(valor) FROM cursos GROUP BY carga_horaria;


Liste os status de matrícula que possuem mais de 1 registro.
SELECT a.descricao FROM status_matricula a JOIN matriculas b ON a.id_status = b.id_status GROUP BY a.id_status, a.descricao HAVING COUNT(b.id_matricula) > 1;

    Questões – Consultas com JOIN

Liste o nome dos alunos e os cursos em que estão matriculados.
SELECT a.nome, b.id_curso FROM alunos a LEFT JOIN matriculas b on a.id_aluno = b.id_aluno;


Liste os alunos, cursos e data da matrícula.
SELECT a.nome, b.id_curso, b.data_matricula FROM alunos a LEFT JOIN matriculas b on a.id_aluno = b.id_aluno;


Mostre os alunos matriculados em cursos com valor acima de R$ 1000,00.
SELECT a.nome, b.id_curso FROM alunos a INNER JOIN matriculas b on a.id_aluno = b.id_aluno WHERE id_curso IN (SELECT id_curso FROM cursos WHERE valor > 1000.00);


Liste a quantidade de alunos por curso.
SELECT a.nome, COUNT(b.id_aluno) FROM cursos a LEFT JOIN matriculas b ON a.id_curso = b.id_curso GROUP BY a.id_curso, a.nome;


Liste os cursos que possuem mais de 1 aluno matriculado.
SELECT a.nome, COUNT(b.id_aluno) FROM cursos a INNER JOIN matriculas b ON a.id_curso = b.id_curso GROUP BY a.id_curso, a.nome HAVING COUNT(b.id_aluno) > 1;



Liste o nome dos alunos, o nome dos cursos e a data da matrícula apenas dos alunos que possuem matrícula cadastrada.
SELECT a.nome, c.nome, b.data_matricula FROM alunos a INNER JOIN matriculas b ON a.id_aluno = b.id_aluno INNER JOIN cursos c ON b.id_curso = c.id_curso;


Liste todos os cursos, incluindo aqueles que ainda não possuem alunos matriculados.
SELECT c.nome, COUNT(m.id_aluno) AS quantidade_alunos FROM cursos c LEFT JOIN matriculas m ON c.id_curso = m.id_curso GROUP BY c.id_curso, c.nome;


Liste todos os alunos, incluindo aqueles que ainda não realizaram matrícula.
SELECT a.nome, m.id_curso, m.data_matricula FROM alunos a LEFT JOIN matriculas m ON a.id_aluno = m.id_aluno;


Liste todos os alunos e todos os cursos, incluindo:
    alunos sem matrícula;
    cursos sem alunos.
SELECT a.nome AS aluno, c.nome AS curso, m.data_matricula FROM alunos a LEFT JOIN matriculas m ON a.id_aluno = m.id_aluno LEFT JOIN cursos c ON m.id_curso = c.id_curso UNION SELECT a.nome, c.nome, m.data_matricula FROM cursos c LEFT JOIN matriculas m ON c.id_curso = m.id_curso RIGHT JOIN alunos a ON m.id_aluno = a.id_aluno;
